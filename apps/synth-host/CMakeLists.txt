cmake_minimum_required(VERSION 3.22)
project(GfSynthHost VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
option(GF_HOST_PIPEWIRE "Build native PipeWire backend when available" ON)
option(GF_HOST_JACK "Build JACK backend when available" ON)
if(GF_HOST_PIPEWIRE)
    pkg_check_modules(PIPEWIRE QUIET IMPORTED_TARGET libpipewire-0.3>=1.4)
else()
    set(PIPEWIRE_FOUND FALSE)
endif()
if(GF_HOST_JACK)
    pkg_check_modules(JACK QUIET IMPORTED_TARGET jack)
else()
    set(JACK_FOUND FALSE)
endif()
pkg_check_modules(SYSTEMD QUIET IMPORTED_TARGET libsystemd)
set(GF_ENGINE_LIBRARIES_ONLY ON CACHE BOOL "Only engine libraries" FORCE)
add_subdirectory(../brain-plugin brain-engines EXCLUDE_FROM_ALL)
add_library(gf_host_core STATIC src/params.cpp src/engine.cpp src/audio.cpp)
target_include_directories(gf_host_core PUBLIC src ../brain-plugin/bench)
target_link_libraries(gf_host_core PUBLIC gf_prophet_preset Threads::Threads)
add_executable(gf-synth-host src/main.cpp src/server.cpp src/backends.cpp src/metadata.cpp)
target_link_libraries(gf-synth-host PRIVATE gf_host_core PkgConfig::JSONC)
target_compile_definitions(gf-synth-host PRIVATE
    GF_HOST_BUILD_TYPE="$<CONFIG>"
    GF_HOST_POLICY="$<IF:$<CONFIG:Release>,O2 -fno-fast-math -ffp-contract=off + FTZ,unoptimized diagnostic build + FTZ>")
if(PIPEWIRE_FOUND)
    target_compile_definitions(gf-synth-host PRIVATE GF_HAVE_PIPEWIRE)
    target_link_libraries(gf-synth-host PRIVATE PkgConfig::PIPEWIRE)
endif()
if(JACK_FOUND)
    target_compile_definitions(gf-synth-host PRIVATE GF_HAVE_JACK)
    target_link_libraries(gf-synth-host PRIVATE PkgConfig::JACK)
endif()
if(SYSTEMD_FOUND)
    target_compile_definitions(gf-synth-host PRIVATE GF_HAVE_SYSTEMD)
    target_link_libraries(gf-synth-host PRIVATE PkgConfig::SYSTEMD)
endif()
include(CheckCXXCompilerFlag)
set(host_flags -O2 -fno-fast-math -ffp-contract=off)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    if(APPLE)
        set(host_cpu -mcpu=apple-m1)
    else()
        set(host_cpu -mcpu=cortex-a76)
    endif()
    check_cxx_compiler_flag("${host_cpu}" HOST_CPU_SUPPORTED)
    if(HOST_CPU_SUPPORTED)
        list(APPEND host_flags ${host_cpu})
    endif()
endif()
foreach(t gf-synth-host gf_host_core gf_prophet_preview gf_prophet_preset gf_preset_codec gf_dx7_6op)
    target_compile_options(${t} PRIVATE "$<$<CONFIG:Release>:${host_flags}>")
endforeach()
target_compile_options(gf_host_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(gf-synth-host PRIVATE -Wall -Wextra -Wpedantic)
message(STATUS "Host: PipeWire=${PIPEWIRE_FOUND}; JACK=${JACK_FOUND}; sd_notify=${SYSTEMD_FOUND}")
include(CTest)
if(BUILD_TESTING)
    add_executable(gf_host_rt_test tests/rt_test.cpp)
    target_link_libraries(gf_host_rt_test PRIVATE gf_host_core)
    target_compile_options(gf_host_rt_test PRIVATE "$<$<CONFIG:Release>:${host_flags}>")
    add_test(NAME host_rt COMMAND gf_host_rt_test)
    add_executable(gf_host_audio_test tests/audio_test.cpp)
    target_link_libraries(gf_host_audio_test PRIVATE gf_host_core)
    target_compile_options(gf_host_audio_test PRIVATE "$<$<CONFIG:Release>:${host_flags}>")
    add_test(NAME host_audio COMMAND gf_host_audio_test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME host_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/contract_test.py $<TARGET_FILE:gf-synth-host>)
    add_test(NAME host_render COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/render_test.py $<TARGET_FILE:gf-synth-host>)
endif()

add_executable(gf-export-metadata src/metadata_export.cpp src/metadata.cpp)
target_link_libraries(gf-export-metadata PRIVATE gf_host_core PkgConfig::JSONC)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/synth-metadata.json"
    COMMAND gf-export-metadata "${CMAKE_CURRENT_BINARY_DIR}/synth-metadata.json"
    DEPENDS gf-export-metadata VERBATIM)
add_custom_target(synth_metadata ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/synth-metadata.json")
if(BUILD_TESTING)
    add_test(NAME metadata_snapshot COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/metadata_test.py
        ${CMAKE_CURRENT_BINARY_DIR}/synth-metadata.json
        ${CMAKE_CURRENT_SOURCE_DIR}/../consola/gui-console/synth-metadata.json)
endif()
if(BUILD_TESTING)
    add_executable(gf_host_dx7_test tests/dx7_test.cpp)
    target_link_libraries(gf_host_dx7_test PRIVATE gf_host_core)
    target_compile_options(gf_host_dx7_test PRIVATE "$<$<CONFIG:Release>:${host_flags}>")
    add_test(NAME host_dx7 COMMAND gf_host_dx7_test)
endif()
