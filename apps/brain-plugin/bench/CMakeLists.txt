# The DSP libraries must receive the same floating-point and optimization policy as the runner.
include(CheckCXXCompilerFlag)
set(GF_BENCH_OPT "O2" CACHE STRING "Benchmark Release optimization (O2 or O3)")
set_property(CACHE GF_BENCH_OPT PROPERTY STRINGS O2 O3)
if(NOT GF_BENCH_OPT MATCHES "^O[23]$")
    message(FATAL_ERROR "GF_BENCH_OPT must be O2 or O3")
endif()
set(GF_BENCH_FLAGS "-${GF_BENCH_OPT}" -fno-fast-math -ffp-contract=off)
foreach(flag IN LISTS GF_BENCH_FLAGS)
    string(MAKE_C_IDENTIFIER "GF_SUPPORTS_${flag}" flag_test)
    check_cxx_compiler_flag("${flag}" "${flag_test}")
    if(NOT ${flag_test})
        message(FATAL_ERROR "Compiler does not support required benchmark flag ${flag}")
    endif()
endforeach()
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" gf_processor)
if(gf_processor MATCHES "^(aarch64|arm64)$")
    if(APPLE)
        check_cxx_compiler_flag("-mcpu=apple-m1" GF_SUPPORTS_APPLE_M1)
        if(GF_SUPPORTS_APPLE_M1)
            list(APPEND GF_BENCH_FLAGS -mcpu=apple-m1)
        endif()
    else()
        check_cxx_compiler_flag("-mcpu=cortex-a76" GF_SUPPORTS_CORTEX_A76)
        if(GF_SUPPORTS_CORTEX_A76)
            list(APPEND GF_BENCH_FLAGS -mcpu=cortex-a76)
        else()
            check_cxx_compiler_flag("-march=armv8.2-a+fp16+dotprod" GF_SUPPORTS_ARMV82)
            if(GF_SUPPORTS_ARMV82)
                list(APPEND GF_BENCH_FLAGS -march=armv8.2-a+fp16+dotprod)
            else()
                message(WARNING "No A76/ARMv8.2 tuning supported; using generic aarch64")
            endif()
        endif()
    endif()
endif()
list(JOIN GF_BENCH_FLAGS " " GF_BENCH_FLAGS_TEXT)
message(STATUS "Gemelo benchmark Release flags: ${GF_BENCH_FLAGS_TEXT}")
set(GF_BENCH_REVISION "unknown")
set(GF_BENCH_DIRTY "unknown")
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE GF_BENCH_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE gf_status OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(gf_status)
        set(GF_BENCH_DIRTY "true")
    else()
        set(GF_BENCH_DIRTY "false")
    endif()
endif()
# Include uncommitted sources in provenance; git revision alone is insufficient in a sandbox.
file(GLOB_RECURSE gf_snapshot_sources LIST_DIRECTORIES false
    "${PROJECT_SOURCE_DIR}/src/*.cpp" "${PROJECT_SOURCE_DIR}/src/*.h"
    "${GF_FIRMWARE}/*.cpp" "${GF_FIRMWARE}/*.h" "${GF_BRIDGE}/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.in")
list(APPEND gf_snapshot_sources "${PROJECT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT gf_snapshot_sources)
set(gf_source_manifest "")
foreach(source IN LISTS gf_snapshot_sources)
    file(SHA256 "${source}" digest)
    file(RELATIVE_PATH relative_source "${PROJECT_SOURCE_DIR}/../.." "${source}")
    string(APPEND gf_source_manifest "${digest}  ${relative_source}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-sha256.txt" "${gf_source_manifest}")
string(SHA256 GF_BENCH_SOURCE_HASH "${gf_source_manifest}")
configure_file(build_info.h.in "${CMAKE_CURRENT_BINARY_DIR}/build_info.h" @ONLY)
add_executable(gf_engine_bench engine_bench.cpp)
target_include_directories(gf_engine_bench PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(gf_engine_bench PRIVATE gf_prophet_preset)
target_compile_definitions(gf_engine_bench PRIVATE GF_BENCH_BUILD_TYPE="$<CONFIG>")
# Keep the executable alongside existing JUCE-free test binaries for a stable CLI.
set_target_properties(gf_engine_bench PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(target gf_engine_bench gf_prophet_preview gf_prophet_preset gf_preset_codec)
    target_compile_options(${target} PRIVATE "$<$<CONFIG:Release>:${GF_BENCH_FLAGS}>")
endforeach()
add_test(NAME engine_bench_self_test COMMAND gf_engine_bench --self-test)
add_test(NAME engine_bench_four_part_smoke COMMAND gf_engine_bench --seconds 0.1 --block 128
    --engine four-part --json "${CMAKE_CURRENT_BINARY_DIR}/smoke/results.json" --threads 1)
